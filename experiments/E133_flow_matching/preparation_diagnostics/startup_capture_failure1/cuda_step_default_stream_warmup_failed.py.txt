"""Graph-captured random-time velocity regression with pre-Adam guards."""
import torch
from .batches import inputs_only
from .augmentation import axis_swap
from .flow import path,objective
from .captured_projection import CapturedProjection

def augmented(batch):
    # Same public/target transposition as E132, without unused port tensors.
    return axis_swap(batch)

class CapturedStep:
    def __init__(self,model,optimizer,sample,phase='fm',updates_authorized=False):
        if updates_authorized:
            from .control import assert_training_authorized
            assert_training_authorized(phase)
        self.updates_authorized=updates_authorized;self.model=model;self.optimizer=optimizer
        self.inputs={k:v.clone() for k,v in sample.items()}
        self.inputs['axis_swap']=torch.zeros(len(sample['valid']),dtype=torch.bool,device='cuda')
        rng=torch.cuda.get_rng_state_all();stream=torch.cuda.Stream();stream.wait_stream(torch.cuda.current_stream())
        def calculation():
            optimizer.zero_grad(set_to_none=False)
            batch=augmented(self.inputs);valid=batch['valid'];z1=model.standardize(batch['target'])*valid[...,None]
            self.z0=torch.randn_like(z1)*valid[...,None]
            self.t=torch.rand(z1.shape[0],device=z1.device)
            zt,target=path(self.z0,z1,self.t)
            out=model(inputs_only(batch),zt,self.t)
            self.per_layout=objective(out,target,valid)
            self.values={k:v.mean() for k,v in self.per_layout.items()}
            self.values['total'].backward()
            self.norm=torch.nn.utils.clip_grad_norm_(model.parameters(),1.,error_if_nonfinite=False,foreach=True)
            self.max_case=self.per_layout['total'].amax()
            self.finite=torch.isfinite(torch.stack(list(self.values.values()))).all()&torch.isfinite(out).all()&torch.isfinite(self.norm)&torch.isfinite(self.per_layout['total']).all()
        # Initialize the autograd CUDA worker/stream state outside capture.
        # A process whose first backward is on a side stream can otherwise
        # create a legacy-stream dependency during capture on this Windows build.
        calculation()
        torch.cuda.synchronize()
        optimizer.zero_grad(set_to_none=True)
        stream.wait_stream(torch.cuda.current_stream())
        with torch.cuda.stream(stream):
            for _ in range(3):calculation()
        torch.cuda.current_stream().wait_stream(stream);torch.cuda.synchronize()
        self.graph=torch.cuda.CUDAGraph()
        with torch.cuda.graph(self.graph,stream=stream):calculation()
        torch.cuda.synchronize();torch.cuda.set_rng_state_all(rng)
        self.projection=CapturedProjection(model)
    def backward_only(self):
        self.graph.replay()
        if not bool(self.finite&(self.max_case<100.)&(self.values['total']<100.)&(self.norm<10000.)):
            raise RuntimeError('FM numerical guard rejected before AdamW')
        return self.values,self.norm
    def replay(self):
        if not self.updates_authorized:raise PermissionError('Preparation only')
        values,norm=self.backward_only();self.projection.before();self.optimizer.step();self.projection.after()
        return values,norm
